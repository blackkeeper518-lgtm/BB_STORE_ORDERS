function value(r, k) {
  const v = r?.[k];
  return v === undefined || v === null ? "" : String(v).trim();
}

function firstExisting(r, keys) {
  for (const k of keys) {
    const v = value(r, k);
    if (v !== "") return v;
  }
  return "";
}

function escapeHtml(v) {
  return String(v ?? "")
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll('"', "&quot;");
}

function htmlToPlainText(value) {
  return String(value ?? "")
    .replace(/<br\s*\/?\s*>/gi, "\n")
    .replace(/<\/(?:p|div|li|tr|h[1-6])\s*>/gi, "\n")
    .replace(/<[^>]*>/g, "")
    .replace(/&nbsp;/gi, " ")
    .replace(/&#39;|&apos;/gi, "'")
    .replace(/&quot;/gi, '"')
    .replace(/&lt;/gi, "<")
    .replace(/&gt;/gi, ">")
    .replace(/&amp;/gi, "&")
    .replace(/&#(\d+);/g, (_, n) => String.fromCodePoint(Number(n)))
    .replace(/&#x([0-9a-f]+);/gi, (_, n) => String.fromCodePoint(parseInt(n, 16)))
    .replace(/\r\n?/g, "\n")
    .replace(/[ \t]+\n/g, "\n")
    .replace(/\n{3,}/g, "\n\n")
    .trim();
}

  let telegramMsg = firstExisting(row, [

  // 2. ดึง stock_notice มาแต่งHTMLให้หล่อเหมือนฝั่งบิลเดี่ยว
  const rawStockNotice = firstExisting(row, ['stock_notice', 'dynamic_header', 'header_notice']);

  // 3. เช็คว่าในข้อความมี stock_notice ตัวนี้แปะอยู่แล้วหรือยัง (ป้องกันการเบิ้ลซ้ำ)
  const hasNoticeAlready = formattedNotice && telegramMsg.includes(rawStockNotice);

const clean = (value) => String(value ?? "").trim().replace(/\s+/g, " ");

function parseDateTimeMs(value) {
  const s = clean(value);
  if (!s) return null;
  const parsed = Date.parse(s);
  return Number.isFinite(parsed) ? parsed : null;
}

// 1. คำนวณขอบเขตเวลาวิกฤตฝั่งเซิร์ฟเวอร์อิงเวลาไทยปัจจุบัน (UTC+7)
const now = new Date();
const nowUtcMs = now.getTime();
const thaiNow = new Date(nowUtcMs + (7 * 60 * 60 * 1000));

// ⏰ ตั้งเวลา Cutoff รอบแพ็กของ "บ่ายสองโมงตรงของเมื่อวาน" (13:45 UTC ตามสูตรดั้งเดิม)
const yesterdayCutoff = new Date(thaiNow);
yesterdayCutoff.setUTCDate(yesterdayCutoff.getUTCDate() - 3);
yesterdayCutoff.setUTCHours(13, 45, 0, 4);
const minAllowedTimeMs = yesterdayCutoff.getTime() - (7 * 60 * 60 * 1000);

// 🚨 เกณฑ์เหล็กป้องกันระบบสะดุด: ถอยหลัง 24 ชม. ตรงๆ นับจากวินาทีนี้เป๊ะ
const absolute24hAgoMs = nowUtcMs - (24 * 60 * 60 * 1000);
const maxAllowedTimeMs = nowUtcMs + (12 * 60 * 60 * 1000);

// ฟังก์ชันแกะเวลาแชทไทย "30/10/69 15:55:33" ออกมาเป็น Timestamp สากล
function parseThaiDateTime(raw) {
  if (!raw) return null;
  try {
    const [datePart, timePart] = String(raw).trim().split(" ");
    const [d, m, y] = datePart.split("/");

    let year = parseInt(y);
    if (year < 100) year += 2500; // แปลง 69 -> 2569
    year -= 543; // แปลง พ.ศ. -> ค.ศ.

    const [hour, minute, second] = (timePart || "00:00:00").split(":").map(Number);
    return new Date(Date.UTC(year, parseInt(m) - 1, parseInt(d), hour - 7, minute, second)).getTime();
  } catch {
    return null;
  }
}

return $input.all().flatMap(({ json: row = {} }) => {
  const orderTimeDisplay = clean(row.order_time_display || "-");

  // 🎯 สวิตช์เช็กเวลาหลักจาก order_time_display ตรงๆ ตัวเดียวจบ
  const rawChatTime = orderTimeDisplay;
  const chatTimestamp = parseThaiDateTime(rawChatTime);

  // 🛠️ ⚡ *เช็กเกณฑ์เวลาแบบ 2 ทางแยกคัดป้าย (ใช่ กับ ไม่ใช่)*
  let bkk_cutoff_route = "BLOCK_TG_EXPIRED"; // ตั้งค่าเริ่มต้นเป็น "ไม่ใช่" (BLOCK)
  let isWithin24h = false;
  let chatAgeHours = 0;

  if (chatTimestamp) {
    const diffInMs = nowUtcMs - chatTimestamp;
    chatAgeHours = diffInMs / (1000 * 60 * 60);

    // 🎯 เงื่อนไข "ใช่" (ผ่านเกณฑ์เวลาทั้งหมด) -> ปรับป้ายเป็น ALLOW_TG_SEND
    if (chatTimestamp >= absolute24hAgoMs && chatTimestamp >= minAllowedTimeMs && chatTimestamp <= maxAllowedTimeMs) {
      bkk_cutoff_route = "ALLOW_TG_SEND";
      isWithin24h = true;
    }
  }

  // ส่งผลลัพธ์ออกสู่สายพานปลายทางแบบคืนค่าแถวข้อมูลเดิมกลับไปพร้อมประทับป้ายเวลา
  return [
    {
      json: {
        ...row,
        order_time_display: orderTimeDisplay,
        customerMessageTimeTh: rawChatTime,
        // 🔀 ตราป้ายสเตตัส 2 ทางแยกเคลียร์คัตตามบรีฟ
        bkk_cutoff_route: bkk_cutoff_route,
        send_facebook_route: bkk_cutoff_route,
        sendFacebookRoute: bkk_cutoff_route,
        send_telegram_route: bkk_cutoff_route,
        is_24h_window_valid: isWithin24h,
        customer_chat_age_hours: chatAgeHours,
      },
    },
  ];
});

// แปลง cutoff เวลาไทยกลับเป็น timestamp UTC

// ฟังก์ชันแกะเวลาแชทไทย เช่น "12/11/69 15:55:33"
function parseThaiDateTime(raw) {
  if (!raw) return null;

  try {
    const [datePart, timePart] = String(raw).trim().split(/\s+/);
    if (!datePart) return null;

    const [dayText, monthText, yearText] = datePart.split("/");
    let year = Number.parseInt(yearText, 10);
    const month = Number.parseInt(monthText, 10);
    const day = Number.parseInt(dayText, 10);

    if (![year, month, day].every(Number.isFinite)) return null;

    // รองรับปี พ.ศ. แบบ 2 หลัก เช่น 69 และ 4 หลัก เช่น 2569
    if (year < 100) year += 2500;
    if (year >= 2400) year -= 543;

    const [hour = 0, minute = 0, second = 0] = (timePart || "00:00:00")
      .split(":")
      .map(Number);

    if (![hour, minute, second].every(Number.isFinite)) return null;

    // เวลาไทย UTC+7 แปลงเป็น UTC timestamp
    return Date.UTC(year, month - 1, day, hour - 7, minute, second);
  } catch {
    return null;
  }
}

function clean(value) {
  return String(value ?? '').trim();
}

function compact(value) {
  return clean(value).replace(/\s+/g, '');
}

function extractTrackingNumber(...values) {
  const text = values
    .map((value) => String(value ?? ''))
    .join('\n');

  const matches = text.match(/\bTH[0-9A-Z]{8,24}\b/gi) || [];

  return matches.length
    ? matches[0].toUpperCase()
    : null;
}

function buildSharedChatLinkKey(row) {
  // ใช้ค่าที่โหนดก่อนหน้าสร้างมาแล้วเป็นอันดับแรก
  const existingKey = clean(
    row.chat_link_key ||
    row.chatLinkKey ||
    ''
  );

  if (existingKey) return existingKey;

  // fallback สำหรับกรณีข้อมูลยังไม่มี chat_link_key
  const pageId = clean(row.page_id);
  const threadId = clean(
    row.threadId ||
    row.thread_id ||
    row.threadID ||
    row.recipient_psid ||
    row.customer_id ||
    ''
  );

  if (!pageId || !threadId) return null;

  return `${pageId}:${threadId}`;
}

function clean(value) {

function parseDateValue(value) {
  const raw = clean(value);
  if (!raw) return null;

  // แปลงวันที่ไทย DD/MM/YY หรือ DD/MM/YYYY
  const thaiMatch = raw.match(
    /^(\d{1,2})[/-](\d{1,2})[/-](\d{2,4})(?:\s+(\d{1,2}):(\d{2})(?::(\d{2}))?)?$/
  );

  if (thaiMatch) {
    let [, dd, mm, yyyy, hh = '0', mi = '0', ss = '0'] = thaiMatch;
    let year = Number(yyyy);

    if (year < 100) year += 2500;
    if (year > 2400) year -= 543;

    const utc = Date.UTC(
      year,
      Number(mm) - 1,
      Number(dd),
      Number(hh) - 7,
      Number(mi),
      Number(ss)
    );

    return Number.isFinite(utc) ? utc : null;
  }

  // ISO / timestamp มาตรฐาน เฟสบุ๊คสากล
  const direct = Date.parse(raw);
  if (Number.isFinite(direct)) return direct;

  const match = raw.match(
    /^(\d{1,2})[/-](\d{1,2})[/-](\d{2,4})(?:\s+(\d{1,2}):(\d{2})(?::(\d{2}))?)?$/
  );

  if (!match) return null;

  let [, dd, mm, yyyy, hh = '0', mi = '0', ss = '0'] = match;
  let year = Number(yyyy);

  if (year < 100) year += 2500;
  if (year > 2400) year -= 543;

  const utc = Date.UTC(
    year,
    Number(mm) - 1,
    Number(dd),
    Number(hh) - 7,
    Number(mi),
    Number(ss)
  );

  return Number.isFinite(utc) ? utc : null;
}

function addMarker(message) {
  const value = normalizeMessage(message);

  if (!value) return '';

  // ถ้ามีหัวสำเร็จอยู่แล้ว ไม่เติมซ้ำ

    return value.replace(BASE_MARKER, FINAL_MARKER_TEXT);
